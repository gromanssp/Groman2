import { computed, Injectable, signal } from '@angular/core';
import { readStorage, removeStorage, writeStorage } from './storage';

export type UserRole = 'admin' | 'editor' | 'user';

export interface UserProfile {
  uid: string;
  email: string;
  displayName: string;
  role: UserRole;
  createdAt: string;
}

const USER_KEY = 'groman-user';

/**
 * Demo authentication backed by `localStorage`. Swap the three async methods
 * for real HTTP calls - the signal surface stays the same.
 */
@Injectable({ providedIn: 'root' })
export class AuthService {
  private readonly _user = signal<UserProfile | null>(readStorage<UserProfile | null>(USER_KEY, null));

  readonly user = this._user.asReadonly();
  readonly profile = this._user.asReadonly();
  readonly isAuthenticated = computed(() => this._user() !== null);
  readonly role = computed<UserRole | null>(() => this._user()?.role ?? null);
  readonly initials = computed(() => {
    const name = this._user()?.displayName ?? '';
    return name
      .split(' ')
      .filter(Boolean)
      .slice(0, 2)
      .map(part => part[0]!.toUpperCase())
      .join('');
  });

  async register(email: string, password: string, displayName: string): Promise<void> {
    this.persist(this.createProfile(email, displayName));
  }

  async login(email: string, password: string): Promise<void> {
    const stored = readStorage<UserProfile | null>(USER_KEY, null);
    this.persist(stored ?? this.createProfile(email, email.split('@')[0]!));
  }

  async loginWithGoogle(): Promise<void> {
    await this.login('user@gmail.com', '');
  }

  logout(): void {
    this._user.set(null);
    removeStorage(USER_KEY);
  }

  private createProfile(email: string, displayName: string): UserProfile {
    return {
      uid: crypto.randomUUID(),
      email,
      displayName,
      role: 'user',
      createdAt: new Date().toISOString()
    };
  }

  private persist(profile: UserProfile): void {
    this._user.set(profile);
    writeStorage(USER_KEY, profile);
  }
}
