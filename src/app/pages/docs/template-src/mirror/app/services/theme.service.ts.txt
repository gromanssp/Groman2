import { computed, effect, Injectable } from '@angular/core';
import { persistedSignal } from './storage';

export type NavbarMode = 'dark' | 'colored';
export type ColorScheme = 'dark' | 'light';

export interface ThemeConfig {
  key: string;
  name: string;
  primary: string;
  secondary: string;
  gradientHover: string;
  navbarBg: string;
  navbarText: string;
  navbarMode: NavbarMode;
}

const COLORS = [
  { key: 'indigo', name: 'Indigo', primary: '#6366f1', secondary: '#8b5cf6', gradientHover: '#818cf8, #a78bfa' },
  { key: 'blue', name: 'Blue', primary: '#3b82f6', secondary: '#2563eb', gradientHover: '#60a5fa, #3b82f6' },
  { key: 'green', name: 'Green', primary: '#10b981', secondary: '#059669', gradientHover: '#34d399, #10b981' },
  { key: 'purple', name: 'Purple', primary: '#a855f7', secondary: '#7c3aed', gradientHover: '#c084fc, #a78bfa' },
  { key: 'red', name: 'Red', primary: '#ef4444', secondary: '#dc2626', gradientHover: '#f87171, #ef4444' },
  { key: 'teal', name: 'Teal', primary: '#14b8a6', secondary: '#0d9488', gradientHover: '#2dd4bf, #14b8a6' }
] as const;

const THEMES: readonly ThemeConfig[] = [
  ...COLORS.map(c => ({ ...c, navbarBg: '', navbarText: '', navbarMode: 'dark' as const })),
  ...COLORS.map(c => ({
    ...c,
    key: `${c.key}-nav`,
    navbarBg: `linear-gradient(135deg, ${c.primary} 0%, ${c.secondary} 100%)`,
    navbarText: '#ffffff',
    navbarMode: 'colored' as const
  }))
];

@Injectable({ providedIn: 'root' })
export class ThemeService {
  readonly themes = THEMES;
  readonly darkNavbarThemes = THEMES.filter(t => t.navbarMode === 'dark');
  readonly coloredNavbarThemes = THEMES.filter(t => t.navbarMode === 'colored');

  private readonly _themeKey = persistedSignal<string>('groman-theme', 'indigo');
  private readonly _colorScheme = persistedSignal<ColorScheme>('groman-color-scheme', 'dark');

  readonly themeKey = this._themeKey.asReadonly();
  readonly colorScheme = this._colorScheme.asReadonly();
  readonly isLight = computed(() => this._colorScheme() === 'light');

  readonly activeTheme = computed<ThemeConfig>(
    () => THEMES.find(t => t.key === this._themeKey()) ?? THEMES[0]
  );

  constructor() {
    effect(() => this.applyAccent(this.activeTheme()));
    effect(() => document.documentElement.setAttribute('data-theme', this._colorScheme()));
  }

  setTheme(key: string): void {
    if (THEMES.some(t => t.key === key)) {
      this._themeKey.set(key);
    }
  }

  setColorScheme(scheme: ColorScheme): void {
    this._colorScheme.set(scheme);
  }

  toggleColorScheme(): void {
    this._colorScheme.update(scheme => (scheme === 'dark' ? 'light' : 'dark'));
  }

  private applyAccent(theme: ThemeConfig): void {
    const style = document.documentElement.style;
    style.setProperty('--accent-primary', theme.primary);
    style.setProperty('--accent-secondary', theme.secondary);
    style.setProperty('--accent-gradient', `linear-gradient(135deg, ${theme.primary} 0%, ${theme.secondary} 100%)`);

    const [hoverStart, hoverEnd] = theme.gradientHover.split(', ');
    style.setProperty('--accent-gradient-hover', `linear-gradient(135deg, ${hoverStart} 0%, ${hoverEnd} 100%)`);

    const navbarVars: Record<string, string> = {
      '--navbar-bg': theme.navbarBg,
      '--navbar-text': theme.navbarText,
      '--navbar-text-secondary': 'rgba(255, 255, 255, 0.8)',
      '--navbar-text-muted': 'rgba(255, 255, 255, 0.55)',
      '--navbar-hover-bg': 'rgba(255, 255, 255, 0.15)',
      '--navbar-border': '1px solid rgba(255, 255, 255, 0.18)',
      '--navbar-search-bg': 'rgba(255, 255, 255, 0.15)',
      '--navbar-search-border': 'rgba(255, 255, 255, 0.25)'
    };

    for (const [name, value] of Object.entries(navbarVars)) {
      if (theme.navbarBg) {
        style.setProperty(name, value);
      } else {
        style.removeProperty(name);
      }
    }
  }
}
