import { Injectable, signal } from '@angular/core';
import { persistedSignal } from './storage';

@Injectable({ providedIn: 'root' })
export class SidebarService {
  /** Desktop collapse survives reloads; the mobile drawer always starts closed. */
  private readonly _collapsed = persistedSignal('groman-sidebar-collapsed', false);
  private readonly _mobileOpen = signal(false);

  readonly collapsed = this._collapsed.asReadonly();
  readonly mobileOpen = this._mobileOpen.asReadonly();

  toggle(): void {
    this._collapsed.update(value => !value);
  }

  collapse(): void {
    this._collapsed.set(true);
  }

  expand(): void {
    this._collapsed.set(false);
  }

  toggleMobile(): void {
    this._mobileOpen.update(value => !value);
  }

  openMobile(): void {
    this._mobileOpen.set(true);
  }

  closeMobile(): void {
    this._mobileOpen.set(false);
  }
}
