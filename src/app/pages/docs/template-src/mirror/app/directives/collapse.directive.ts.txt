import { booleanAttribute, Directive, effect, ElementRef, inject, input } from '@angular/core';

/**
 * Animates a block between collapsed and expanded using its measured height.
 *
 * The effect only writes styles - no component state changes - so it is safe to
 * run outside change detection.
 */
@Directive({
  selector: '[appCollapse]',
  host: {
    '[style.overflow]': '"hidden"',
    '[style.transition]': '"max-height 0.3s ease, opacity 0.3s ease"'
  }
})
export class CollapseDirective {
  readonly isCollapsed = input(false, { alias: 'appCollapse', transform: booleanAttribute });

  private readonly host = inject<ElementRef<HTMLElement>>(ElementRef);

  constructor() {
    effect(() => (this.isCollapsed() ? this.collapse() : this.expand()));
  }

  private collapse(): void {
    const el = this.host.nativeElement;
    el.style.maxHeight = `${el.scrollHeight}px`;
    requestAnimationFrame(() => {
      el.style.maxHeight = '0';
      el.style.opacity = '0';
    });
  }

  private expand(): void {
    const el = this.host.nativeElement;
    el.style.maxHeight = `${el.scrollHeight}px`;
    el.style.opacity = '1';
    el.addEventListener('transitionend', () => (el.style.maxHeight = 'none'), { once: true });
  }
}
