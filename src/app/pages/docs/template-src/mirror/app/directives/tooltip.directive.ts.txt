import { DestroyRef, Directive, ElementRef, inject, input } from '@angular/core';
import { UiPlacement } from '../shared/ui.types';

/**
 * Text tooltip rendered into `document.body`, so it is never clipped by an
 * `overflow: hidden` ancestor (the collapsed sidebar, table cells, cards).
 */
@Directive({
  selector: '[appTooltip]',
  host: {
    '(mouseenter)': 'show()',
    '(mouseleave)': 'hide()',
    '(focusin)': 'show()',
    '(focusout)': 'hide()'
  }
})
export class TooltipDirective {
  readonly tooltipText = input('', { alias: 'appTooltip' });
  readonly tooltipPosition = input<UiPlacement>('top');

  private readonly host = inject<ElementRef<HTMLElement>>(ElementRef);
  private tooltipEl: HTMLElement | null = null;

  constructor() {
    inject(DestroyRef).onDestroy(() => this.hide());
  }

  protected show(): void {
    const text = this.tooltipText();
    if (!text || this.tooltipEl) return;

    const position = this.tooltipPosition();
    const element = document.createElement('div');
    element.className = `app-tooltip tooltip-${position}`;
    element.setAttribute('role', 'tooltip');
    element.textContent = text;
    document.body.appendChild(element);
    this.tooltipEl = element;

    const rect = this.host.nativeElement.getBoundingClientRect();
    const gap = 8;
    const { top, left, transform } = this.place(position, rect, gap);

    element.style.top = `${top}px`;
    element.style.left = `${left}px`;
    element.style.transform = transform;
    element.style.opacity = '1';
  }

  protected hide(): void {
    this.tooltipEl?.remove();
    this.tooltipEl = null;
  }

  private place(position: UiPlacement, rect: DOMRect, gap: number) {
    switch (position) {
      case 'bottom':
        return { top: rect.bottom + gap, left: rect.left + rect.width / 2, transform: 'translate(-50%, 0) scale(1)' };
      case 'left':
        return { top: rect.top + rect.height / 2, left: rect.left - gap, transform: 'translate(-100%, -50%) scale(1)' };
      case 'right':
        return { top: rect.top + rect.height / 2, left: rect.right + gap, transform: 'translate(0, -50%) scale(1)' };
      default:
        return { top: rect.top - gap, left: rect.left + rect.width / 2, transform: 'translate(-50%, -100%) scale(1)' };
    }
  }
}
