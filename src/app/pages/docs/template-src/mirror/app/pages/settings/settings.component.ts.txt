import { ChangeDetectionStrategy, Component, inject } from '@angular/core';
import { UiSwitchComponent } from '../../shared/components/ui-switch/ui-switch.component';
import { ThemeService } from '../../services/theme.service';
import { ToastService } from '../../services/toast.service';

@Component({
  selector: 'app-settings',
  templateUrl: './settings.component.html',
  styleUrl: './settings.component.css',
  imports: [UiSwitchComponent],
  changeDetection: ChangeDetectionStrategy.OnPush
})
export class SettingsComponent {
  private readonly theme = inject(ThemeService);
  private readonly toasts = inject(ToastService);

  protected readonly darkNavbarThemes = this.theme.darkNavbarThemes;
  protected readonly coloredNavbarThemes = this.theme.coloredNavbarThemes;
  protected readonly selectedTheme = this.theme.themeKey;
  protected readonly isLight = this.theme.isLight;

  protected selectTheme(key: string): void {
    this.theme.setTheme(key);
  }

  protected onSchemeChange(light: boolean): void {
    this.theme.setColorScheme(light ? 'light' : 'dark');
    this.toasts.info(`${light ? 'Light' : 'Dark'} mode enabled`);
  }
}
