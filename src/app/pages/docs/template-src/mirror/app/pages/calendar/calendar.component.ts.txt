import { ChangeDetectionStrategy, Component, computed, inject, signal } from '@angular/core';
import { DatePipe } from '@angular/common';
import { form, required, submit } from '@angular/forms/signals';
import { EmptyStateComponent } from '../../shared/components/empty-state/empty-state.component';
import { ModalComponent } from '../../shared/components/modal/modal.component';
import { UiInputComponent } from '../../shared/components/ui-input/ui-input.component';
import { UiSelectComponent } from '../../shared/components/ui-select/ui-select.component';
import { ToastService } from '../../services/toast.service';

export interface CalendarEvent {
  title: string;
  date: Date;
  color: string;
}

export interface CalendarDay {
  date: number;
  month: number;
  year: number;
  isCurrentMonth: boolean;
  isToday: boolean;
  isSelected: boolean;
  events: CalendarEvent[];
}

const EVENT_COLORS = [
  { value: 'var(--accent-primary)', label: 'Primary' },
  { value: 'var(--success)', label: 'Success' },
  { value: 'var(--warning)', label: 'Warning' },
  { value: 'var(--danger)', label: 'Danger' },
  { value: 'var(--info)', label: 'Info' }
];

@Component({
  selector: 'app-calendar',
  templateUrl: './calendar.component.html',
  styleUrl: './calendar.component.css',
  imports: [DatePipe, ModalComponent, EmptyStateComponent, UiInputComponent, UiSelectComponent],
  changeDetection: ChangeDetectionStrategy.OnPush
})
export class CalendarComponent {
  private readonly toasts = inject(ToastService);

  protected readonly weekDays = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'];
  protected readonly eventColors = EVENT_COLORS;

  private readonly cursor = signal(new Date());
  protected readonly selectedDate = signal<Date | null>(null);
  protected readonly events = signal<CalendarEvent[]>([
    { title: 'Team Standup', date: new Date(2026, 2, 2), color: 'var(--accent-primary)' },
    { title: 'Sprint Review', date: new Date(2026, 2, 6), color: 'var(--success)' },
    { title: 'Product Launch', date: new Date(2026, 2, 14), color: 'var(--danger)' },
    { title: 'Design Workshop', date: new Date(2026, 2, 14), color: 'var(--accent-secondary)' },
    { title: 'Client Meeting', date: new Date(2026, 2, 18), color: 'var(--warning)' },
    { title: 'Code Freeze', date: new Date(2026, 2, 20), color: 'var(--danger)' },
    { title: 'Team Lunch', date: new Date(2026, 2, 25), color: 'var(--success)' },
    { title: 'Quarterly Review', date: new Date(2026, 2, 28), color: 'var(--accent-primary)' }
  ]);

  protected readonly showCreateModal = signal(false);

  protected readonly eventModel = signal({
    title: '',
    date: '',
    color: 'var(--accent-primary)'
  });

  protected readonly eventForm = form(this.eventModel, path => {
    required(path.title, { message: 'Event title is required' });
    required(path.date, { message: 'Pick a date' });
  });

  protected readonly monthYear = computed(() =>
    this.cursor().toLocaleDateString('en-US', { month: 'long', year: 'numeric' })
  );

  protected readonly selectedDateEvents = computed(() => {
    const selected = this.selectedDate();
    return selected ? this.events().filter(event => isSameDay(event.date, selected)) : [];
  });

  /** Six-week grid, rebuilt whenever the cursor, selection or events change. */
  protected readonly weeks = computed<CalendarDay[][]>(() => {
    const cursor = this.cursor();
    const year = cursor.getFullYear();
    const month = cursor.getMonth();
    const firstWeekday = new Date(year, month, 1).getDay();
    const daysInMonth = new Date(year, month + 1, 0).getDate();
    const daysInPrevMonth = new Date(year, month, 0).getDate();

    const days: CalendarDay[] = [
      ...Array.from({ length: firstWeekday }, (_, i) =>
        this.buildDay(daysInPrevMonth - firstWeekday + 1 + i, month - 1, year, false)
      ),
      ...Array.from({ length: daysInMonth }, (_, i) => this.buildDay(i + 1, month, year, true))
    ];
    while (days.length < 42) {
      days.push(this.buildDay(days.length - firstWeekday - daysInMonth + 1, month + 1, year, false));
    }

    return Array.from({ length: 6 }, (_, i) => days.slice(i * 7, i * 7 + 7));
  });

  protected shiftMonth(delta: number): void {
    this.cursor.update(current => new Date(current.getFullYear(), current.getMonth() + delta, 1));
  }

  protected selectDay(day: CalendarDay): void {
    if (!day.isCurrentMonth) return;
    this.selectedDate.set(new Date(day.year, day.month, day.date));
  }

  protected openCreateModal(): void {
    this.eventModel.set({
      title: '',
      date: toInputDate(this.selectedDate() ?? new Date()),
      color: 'var(--accent-primary)'
    });
    this.showCreateModal.set(true);
  }

  protected createEvent(): void {
    submit(this.eventForm, async () => {
      const { title, date, color } = this.eventModel();
      const [year, month, day] = date.split('-').map(Number);
      this.events.update(list => [...list, { title, color, date: new Date(year!, month! - 1, day!) }]);
      this.showCreateModal.set(false);
      this.toasts.success('Event created', { message: title });
    });
  }

  private buildDay(date: number, month: number, year: number, isCurrentMonth: boolean): CalendarDay {
    const actual = new Date(year, month, date);
    const selected = this.selectedDate();

    return {
      date: actual.getDate(),
      month: actual.getMonth(),
      year: actual.getFullYear(),
      isCurrentMonth,
      isToday: isSameDay(actual, new Date()),
      isSelected: isCurrentMonth && selected !== null && isSameDay(actual, selected),
      events: isCurrentMonth ? this.events().filter(event => isSameDay(event.date, actual)) : []
    };
  }
}

function isSameDay(a: Date, b: Date): boolean {
  return (
    a.getDate() === b.getDate() && a.getMonth() === b.getMonth() && a.getFullYear() === b.getFullYear()
  );
}

function toInputDate(date: Date): string {
  const month = String(date.getMonth() + 1).padStart(2, '0');
  const day = String(date.getDate()).padStart(2, '0');
  return `${date.getFullYear()}-${month}-${day}`;
}
