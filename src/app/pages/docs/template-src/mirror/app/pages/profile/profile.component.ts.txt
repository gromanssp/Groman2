import { ChangeDetectionStrategy, Component, effect, inject, signal, untracked } from '@angular/core';
import { email, form, minLength, required, submit, validate } from '@angular/forms/signals';
import { AvatarComponent } from '../../shared/components/avatar/avatar.component';
import { TabDirective } from '../../shared/components/tabs/tab.directive';
import { TabsComponent } from '../../shared/components/tabs/tabs.component';
import { TimelineComponent, TimelineItem } from '../../shared/components/timeline/timeline.component';
import { UiInputComponent } from '../../shared/components/ui-input/ui-input.component';
import { AuthService } from '../../services/auth.service';
import { ToastService } from '../../services/toast.service';

@Component({
  selector: 'app-profile',
  templateUrl: './profile.component.html',
  styleUrl: './profile.component.css',
  imports: [AvatarComponent, TabsComponent, TabDirective, TimelineComponent, UiInputComponent],
  changeDetection: ChangeDetectionStrategy.OnPush
})
export class ProfileComponent {
  private readonly auth = inject(AuthService);
  private readonly toasts = inject(ToastService);

  protected readonly user = this.auth.profile;

  protected readonly profileModel = signal({ displayName: '', email: '' });
  protected readonly passwordModel = signal({ current: '', next: '', confirm: '' });

  protected readonly profileForm = form(this.profileModel, path => {
    required(path.displayName, { message: 'Display name is required' });
    required(path.email, { message: 'Email is required' });
    email(path.email, { message: 'Enter a valid email address' });
  });

  protected readonly passwordForm = form(this.passwordModel, path => {
    required(path.current, { message: 'Current password is required' });
    required(path.next, { message: 'New password is required' });
    minLength(path.next, 6, { message: 'Password must be at least 6 characters' });
    validate(path.confirm, ({ value, valueOf }) =>
      value() !== valueOf(path.next) ? { kind: 'mismatch', message: 'Passwords do not match' } : undefined
    );
  });

  protected readonly activity: TimelineItem[] = [
    { title: 'Signed in', description: 'From this browser', timestamp: 'Just now', variant: 'success' },
    { title: 'Theme changed', description: 'Switched accent color', timestamp: '3 days ago', variant: 'primary' },
    { title: 'Account created', timestamp: 'Last month', variant: 'secondary' }
  ];

  constructor() {
    // Seed the form from the signed-in user without making the model a
    // derived signal - the user edits it afterwards.
    effect(() => {
      const profile = this.user();
      if (!profile) return;
      untracked(() => this.profileModel.set({ displayName: profile.displayName, email: profile.email }));
    });
  }

  protected saveProfile(): void {
    submit(this.profileForm, async () => {
      this.toasts.success('Profile saved', { message: 'Your changes are stored locally in this demo.' });
    });
  }

  protected savePassword(): void {
    submit(this.passwordForm, async () => {
      this.passwordModel.set({ current: '', next: '', confirm: '' });
      this.toasts.success('Password updated');
    });
  }
}
