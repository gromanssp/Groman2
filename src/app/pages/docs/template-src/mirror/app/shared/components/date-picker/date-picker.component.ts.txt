import { ChangeDetectionStrategy, Component, computed, model, output, signal } from '@angular/core';
import { ClickOutsideDirective } from '../../../directives/click-outside.directive';

export type DatePickerMode = 'single' | 'range' | 'minmax';

export interface DateRange {
  start: Date;
  end: Date;
}

@Component({
  selector: 'app-date-picker',
  templateUrl: './date-picker.component.html',
  styleUrl: './date-picker.component.css',
  imports: [ClickOutsideDirective],
  changeDetection: ChangeDetectionStrategy.OnPush
})
export class DatePickerComponent {
  readonly mode = model<DatePickerMode>('single');
  readonly value = model<Date | null>(null);
  readonly rangeStart = model<Date | null>(null);
  readonly rangeEnd = model<Date | null>(null);
  readonly minDate = model<Date | null>(null);
  readonly maxDate = model<Date | null>(null);
  readonly placeholder = model('Select date');

  readonly dateChange = output<Date>();
  readonly rangeChange = output<DateRange>();

  protected readonly isOpen = signal(false);
  protected readonly weekDays = ['Su', 'Mo', 'Tu', 'We', 'Th', 'Fr', 'Sa'];

  private readonly cursor = signal(startOfMonth(new Date()));
  private rangeSelecting: 'start' | 'end' = 'start';

  protected readonly monthLabel = computed(() =>
    this.cursor().toLocaleString('default', { month: 'long', year: 'numeric' })
  );

  /** Leading `null`s pad the grid so the 1st lands on the right weekday. */
  protected readonly calendarDays = computed<(Date | null)[]>(() => {
    const cursor = this.cursor();
    const year = cursor.getFullYear();
    const month = cursor.getMonth();
    const lastDate = new Date(year, month + 1, 0).getDate();
    const pad = new Date(year, month, 1).getDay();

    return [
      ...Array.from({ length: pad }, () => null),
      ...Array.from({ length: lastDate }, (_, i) => new Date(year, month, i + 1))
    ];
  });

  protected readonly displayValue = computed(() => {
    if (this.mode() !== 'range') {
      const value = this.value();
      return value ? formatDate(value) : '';
    }

    const start = this.rangeStart();
    const end = this.rangeEnd();
    if (start && end) return `${formatDate(start)} - ${formatDate(end)}`;
    if (start) return `${formatDate(start)} - …`;
    return '';
  });

  protected toggleDropdown(): void {
    const next = !this.isOpen();
    this.isOpen.set(next);
    if (next) {
      this.cursor.set(startOfMonth(this.value() ?? this.rangeStart() ?? new Date()));
    }
  }

  protected close(): void {
    this.isOpen.set(false);
  }

  protected shiftMonth(delta: number): void {
    this.cursor.update(current => new Date(current.getFullYear(), current.getMonth() + delta, 1));
  }

  protected selectDay(day: Date | null): void {
    if (!day || this.isDisabled(day)) return;

    if (this.mode() !== 'range') {
      this.value.set(day);
      this.dateChange.emit(day);
      this.close();
      return;
    }

    if (this.rangeSelecting === 'start') {
      this.rangeStart.set(day);
      this.rangeEnd.set(null);
      this.rangeSelecting = 'end';
      return;
    }

    const start = this.rangeStart();
    if (start && day < start) {
      this.rangeEnd.set(start);
      this.rangeStart.set(day);
    } else {
      this.rangeEnd.set(day);
    }
    this.rangeSelecting = 'start';

    const from = this.rangeStart();
    const to = this.rangeEnd();
    if (from && to) {
      this.rangeChange.emit({ start: from, end: to });
      this.close();
    }
  }

  protected isSelected(day: Date | null): boolean {
    if (!day) return false;
    if (this.mode() !== 'range') return isSameDay(day, this.value());
    return isSameDay(day, this.rangeStart()) || isSameDay(day, this.rangeEnd());
  }

  protected isInRange(day: Date | null): boolean {
    const start = this.rangeStart();
    const end = this.rangeEnd();
    if (!day || this.mode() !== 'range' || !start || !end) return false;
    return day > start && day < end;
  }

  protected isRangeStart(day: Date | null): boolean {
    return !!day && this.mode() === 'range' && isSameDay(day, this.rangeStart());
  }

  protected isRangeEnd(day: Date | null): boolean {
    return !!day && this.mode() === 'range' && isSameDay(day, this.rangeEnd());
  }

  protected isDisabled(day: Date | null): boolean {
    if (!day) return true;
    const min = this.minDate();
    const max = this.maxDate();
    if (min && day < stripTime(min)) return true;
    if (max && day > stripTime(max)) return true;
    return false;
  }

  protected isToday(day: Date | null): boolean {
    return !!day && isSameDay(day, new Date());
  }
}

function startOfMonth(date: Date): Date {
  return new Date(date.getFullYear(), date.getMonth(), 1);
}

function stripTime(date: Date): Date {
  return new Date(date.getFullYear(), date.getMonth(), date.getDate());
}

function isSameDay(a: Date | null, b: Date | null): boolean {
  return (
    !!a && !!b &&
    a.getFullYear() === b.getFullYear() &&
    a.getMonth() === b.getMonth() &&
    a.getDate() === b.getDate()
  );
}

function formatDate(date: Date): string {
  const month = String(date.getMonth() + 1).padStart(2, '0');
  const day = String(date.getDate()).padStart(2, '0');
  return `${date.getFullYear()}-${month}-${day}`;
}
