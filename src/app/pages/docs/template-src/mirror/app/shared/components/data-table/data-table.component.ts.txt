import {
  ChangeDetectionStrategy,
  Component,
  computed,
  contentChildren,
  input,
  model,
  numberAttribute,
  output,
  signal
} from '@angular/core';
import { NgTemplateOutlet } from '@angular/common';
import { CellTemplateDirective } from '../../../directives/cell-template.directive';
import { EmptyStateComponent } from '../empty-state/empty-state.component';
import { PaginationComponent } from '../pagination/pagination.component';

export interface TableColumn {
  key: string;
  label: string;
  width?: string;
  sortable?: boolean;
  align?: 'left' | 'center' | 'right';
}

export type TableRow = Record<string, unknown>;
export type SortDirection = 'asc' | 'desc';

export interface SortState {
  key: string;
  direction: SortDirection;
}

/**
 * Table with client-side search, sorting and pagination, all derived with
 * `computed()` so nothing is recalculated unless its inputs change.
 *
 * Per-column rendering comes from `[appCellTemplate]`; loading and empty states
 * are built in.
 */
@Component({
  selector: 'app-data-table',
  templateUrl: './data-table.component.html',
  styleUrl: './data-table.component.css',
  imports: [NgTemplateOutlet, PaginationComponent, EmptyStateComponent],
  changeDetection: ChangeDetectionStrategy.OnPush
})
export class DataTableComponent {
  readonly columns = input<readonly TableColumn[]>([]);
  readonly rows = input<readonly TableRow[]>([]);

  readonly striped = input(false);
  readonly hover = input(true);
  readonly bordered = input(false);
  readonly loading = input(false);

  readonly searchable = input(false);
  readonly searchPlaceholder = input('Search…');
  readonly pageSize = input(0, { transform: numberAttribute });
  readonly emptyMessage = input('No records found');

  readonly sort = model<SortState | null>(null);
  readonly page = model(1);
  readonly rowClick = output<TableRow>();

  private readonly cellTemplates = contentChildren(CellTemplateDirective);
  protected readonly search = signal('');

  protected readonly filtered = computed(() => {
    const term = this.search().trim().toLowerCase();
    if (!term) return this.rows();

    const keys = this.columns().map(column => column.key);
    return this.rows().filter(row =>
      keys.some(key => String(row[key] ?? '').toLowerCase().includes(term))
    );
  });

  protected readonly sorted = computed(() => {
    const sort = this.sort();
    if (!sort) return this.filtered();

    const factor = sort.direction === 'asc' ? 1 : -1;
    return [...this.filtered()].sort((a, b) => compare(a[sort.key], b[sort.key]) * factor);
  });

  protected readonly paginated = computed(() => {
    const size = this.pageSize();
    if (size <= 0) return this.sorted();

    const start = (this.page() - 1) * size;
    return this.sorted().slice(start, start + size);
  });

  protected readonly total = computed(() => this.sorted().length);
  protected readonly isEmpty = computed(() => !this.loading() && this.total() === 0);
  protected readonly skeletonRows = computed(() => Array.from({ length: this.pageSize() || 5 }));

  protected templateFor(key: string) {
    return this.cellTemplates().find(directive => directive.column() === key)?.template ?? null;
  }

  protected toggleSort(column: TableColumn): void {
    if (!column.sortable) return;

    const current = this.sort();
    if (current?.key !== column.key) {
      this.sort.set({ key: column.key, direction: 'asc' });
    } else if (current.direction === 'asc') {
      this.sort.set({ key: column.key, direction: 'desc' });
    } else {
      this.sort.set(null);
    }
    this.page.set(1);
  }

  protected onSearch(event: Event): void {
    this.search.set((event.target as HTMLInputElement).value);
    this.page.set(1);
  }

  protected sortDirectionFor(key: string): SortDirection | null {
    const sort = this.sort();
    return sort?.key === key ? sort.direction : null;
  }
}

function compare(a: unknown, b: unknown): number {
  if (a === b) return 0;
  if (a == null) return -1;
  if (b == null) return 1;
  if (typeof a === 'number' && typeof b === 'number') return a - b;
  return String(a).localeCompare(String(b), undefined, { numeric: true });
}
