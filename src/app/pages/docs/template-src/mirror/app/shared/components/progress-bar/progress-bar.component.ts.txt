import { ChangeDetectionStrategy, Component, computed, input, numberAttribute } from '@angular/core';

export type ProgressVariant = 'primary' | 'success' | 'warning' | 'danger';

@Component({
  selector: 'app-progress-bar',
  templateUrl: './progress-bar.component.html',
  styleUrl: './progress-bar.component.css',
  changeDetection: ChangeDetectionStrategy.OnPush
})
export class ProgressBarComponent {
  readonly value = input(0, { transform: numberAttribute });
  readonly variant = input<ProgressVariant>('primary');
  readonly striped = input(false);
  readonly animated = input(false);
  readonly height = input(8, { transform: numberAttribute });
  readonly label = input('');

  /** Clamped so a stray value never overflows the track. */
  protected readonly percent = computed(() => Math.min(100, Math.max(0, this.value())));
  protected readonly labelInside = computed(() => !!this.label() && this.height() >= 16);
  protected readonly labelOutside = computed(() => !!this.label() && this.height() < 16);
}
