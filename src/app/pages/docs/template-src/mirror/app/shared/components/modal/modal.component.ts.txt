import { ChangeDetectionStrategy, Component, effect, input, model, output } from '@angular/core';
import { AutofocusDirective } from '../../../directives/autofocus.directive';
import { UiSize } from '../../ui.types';

@Component({
  selector: 'app-modal',
  templateUrl: './modal.component.html',
  styleUrl: './modal.component.css',
  imports: [AutofocusDirective],
  host: { '(document:keydown.escape)': 'onEscape()' },
  changeDetection: ChangeDetectionStrategy.OnPush
})
export class ModalComponent {
  /** Two-way: `[(isOpen)]="showModal"` closes the dialog without extra wiring. */
  readonly isOpen = model(false);
  readonly title = input('');
  readonly size = input<UiSize>('md');
  readonly closeOnBackdrop = input(true);
  readonly closed = output<void>();

  constructor() {
    // Prevent the page behind the dialog from scrolling while it is open.
    effect(() => {
      document.body.style.overflow = this.isOpen() ? 'hidden' : '';
    });
  }

  close(): void {
    this.isOpen.set(false);
    this.closed.emit();
  }

  protected onEscape(): void {
    if (this.isOpen()) this.close();
  }

  protected onBackdropClick(event: MouseEvent): void {
    const isBackdrop = (event.target as HTMLElement).classList.contains('modal-backdrop');
    if (isBackdrop && this.closeOnBackdrop()) this.close();
  }
}
