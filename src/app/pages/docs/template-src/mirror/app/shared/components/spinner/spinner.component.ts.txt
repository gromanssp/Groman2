import { ChangeDetectionStrategy, Component, computed, input } from '@angular/core';
import { UI_SIZE_PX, UiSize } from '../../ui.types';

export type SpinnerVariant = 'circular' | 'dots' | 'pulse';

@Component({
  selector: 'app-spinner',
  templateUrl: './spinner.component.html',
  styleUrl: './spinner.component.css',
  host: { role: 'status', '[attr.aria-label]': 'label()' },
  changeDetection: ChangeDetectionStrategy.OnPush
})
export class SpinnerComponent {
  readonly variant = input<SpinnerVariant>('circular');
  readonly size = input<UiSize>('md');
  readonly color = input('var(--accent-primary)');
  readonly label = input('Loading');

  protected readonly sizeInPx = computed(() => UI_SIZE_PX[this.size()]);
  protected readonly dotSize = computed(() => this.sizeInPx() * 0.28);
}
