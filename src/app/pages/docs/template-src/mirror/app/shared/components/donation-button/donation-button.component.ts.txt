import { ChangeDetectionStrategy, Component, computed, inject, input, signal } from '@angular/core';
import { UiStatus } from '../../ui.types';
import { PaymentService } from '../../../services/payment.service';

@Component({
  selector: 'app-donation-button',
  templateUrl: './donation-button.component.html',
  styleUrl: './donation-button.component.css',
  changeDetection: ChangeDetectionStrategy.OnPush
})
export class DonationButtonComponent {
  readonly compact = input(false);

  private readonly payments = inject(PaymentService);

  protected readonly expanded = signal(false);
  protected readonly selectedAmount = signal(0);
  protected readonly customAmount = signal('');
  protected readonly status = signal<UiStatus>('idle');
  protected readonly statusMessage = signal('');

  protected readonly busy = computed(() => this.status() === 'loading');

  protected toggleExpand(): void {
    this.expanded.update(value => !value);
    this.status.set('idle');
    this.statusMessage.set('');
  }

  protected onCustomAmount(event: Event): void {
    this.customAmount.set((event.target as HTMLInputElement).value);
  }

  protected onCustomDonate(): void {
    const amount = Number.parseFloat(this.customAmount());
    if (Number.isFinite(amount) && amount > 0) {
      void this.onDonate(amount);
    }
  }

  protected async onDonate(amount: number): Promise<void> {
    if (amount <= 0) return;

    this.selectedAmount.set(amount);
    this.status.set('loading');
    this.statusMessage.set('');

    try {
      const donationId = await this.payments.createDonation(amount);
      await this.payments.completeDonation(donationId);
      this.status.set('success');
      this.statusMessage.set(`Thank you for your $${amount} donation!`);
    } catch {
      this.status.set('error');
      this.statusMessage.set('Donation failed. Please try again.');
    }
  }
}
