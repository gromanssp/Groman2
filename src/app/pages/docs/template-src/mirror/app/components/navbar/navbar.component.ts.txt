import { ChangeDetectionStrategy, Component, computed, inject, signal } from '@angular/core';
import { Router, RouterLink } from '@angular/router';
import { AvatarComponent } from '../../shared/components/avatar/avatar.component';
import { DropdownComponent } from '../../shared/components/dropdown/dropdown.component';
import { BreadcrumbsComponent } from '../../shared/components/breadcrumbs/breadcrumbs.component';
import { AuthService } from '../../services/auth.service';
import { BreakpointService } from '../../services/breakpoint.service';
import { SidebarService } from '../../services/sidebar.service';
import { ThemeService } from '../../services/theme.service';
import { ToastService } from '../../services/toast.service';
import { NAV_GROUPS } from '../sidebar/nav-items';

interface Notification {
  id: string;
  title: string;
  time: string;
  unread: boolean;
}

@Component({
  selector: 'app-navbar',
  templateUrl: './navbar.component.html',
  styleUrl: './navbar.component.css',
  imports: [RouterLink, AvatarComponent, DropdownComponent, BreadcrumbsComponent],
  host: { '(document:keydown)': 'onGlobalKeydown($event)' },
  changeDetection: ChangeDetectionStrategy.OnPush
})
export class NavbarComponent {
  private readonly sidebar = inject(SidebarService);
  private readonly auth = inject(AuthService);
  private readonly theme = inject(ThemeService);
  private readonly breakpoint = inject(BreakpointService);
  private readonly toasts = inject(ToastService);
  private readonly router = inject(Router);

  protected readonly profile = this.auth.profile;
  protected readonly isAuthenticated = this.auth.isAuthenticated;
  protected readonly isLight = this.theme.isLight;

  protected readonly commandOpen = signal(false);
  protected readonly commandQuery = signal('');

  protected readonly notifications = signal<Notification[]>([
    { id: '1', title: 'New component released: Stepper', time: '2m ago', unread: true },
    { id: '2', title: 'Template ZIP regenerated', time: '1h ago', unread: true },
    { id: '3', title: 'Weekly usage report is ready', time: 'Yesterday', unread: false }
  ]);
  protected readonly unreadCount = computed(() => this.notifications().filter(n => n.unread).length);

  /** Built from the sidebar definition, so a new page shows up in ⌘K for free. */
  private readonly commands = NAV_GROUPS.flatMap(group =>
    group.items.map(item => ({ label: item.label, path: item.path, group: group.title }))
  );

  protected readonly commandResults = computed(() => {
    const query = this.commandQuery().trim().toLowerCase();
    if (!query) return this.commands;
    return this.commands.filter(command => command.label.toLowerCase().includes(query));
  });

  protected toggleSidebar(): void {
    this.breakpoint.isMobile() ? this.sidebar.toggleMobile() : this.sidebar.toggle();
  }

  protected toggleTheme(): void {
    this.theme.toggleColorScheme();
  }

  protected markAllRead(): void {
    this.notifications.update(list => list.map(item => ({ ...item, unread: false })));
    this.toasts.info('All notifications marked as read');
  }

  protected openCommandPalette(): void {
    this.commandQuery.set('');
    this.commandOpen.set(true);
  }

  protected runCommand(path: string): void {
    this.commandOpen.set(false);
    this.router.navigateByUrl(path);
  }

  protected onCommandQuery(event: Event): void {
    this.commandQuery.set((event.target as HTMLInputElement).value);
  }

  protected logout(): void {
    this.auth.logout();
    this.toasts.success('Signed out');
    this.router.navigate(['/auth/login']);
  }

  /** Cmd/Ctrl+K opens the palette, Escape closes it. */
  protected onGlobalKeydown(event: KeyboardEvent): void {
    if ((event.metaKey || event.ctrlKey) && event.key.toLowerCase() === 'k') {
      event.preventDefault();
      this.openCommandPalette();
    } else if (event.key === 'Escape') {
      this.commandOpen.set(false);
    }
  }
}
