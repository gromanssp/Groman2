import { ChangeDetectionStrategy, Component, inject, signal } from '@angular/core';
import { RouterLink, RouterLinkActive } from '@angular/router';
import { TooltipDirective } from '../../directives/tooltip.directive';
import { SidebarService } from '../../services/sidebar.service';
import { NAV_GROUPS } from './nav-items';

@Component({
  selector: 'app-sidebar',
  templateUrl: './sidebar.component.html',
  styleUrl: './sidebar.component.css',
  imports: [RouterLink, RouterLinkActive, TooltipDirective],
  changeDetection: ChangeDetectionStrategy.OnPush
})
export class SidebarComponent {
  private readonly sidebar = inject(SidebarService);

  protected readonly collapsed = this.sidebar.collapsed;
  protected readonly groups = NAV_GROUPS;

  /** Titles of groups the user folded away. */
  private readonly foldedGroups = signal<ReadonlySet<string>>(new Set());

  protected isFolded(title: string): boolean {
    return this.foldedGroups().has(title);
  }

  protected toggleGroup(title: string): void {
    this.foldedGroups.update(folded => {
      const next = new Set(folded);
      next.has(title) ? next.delete(title) : next.add(title);
      return next;
    });
  }

  protected toggle(): void {
    this.sidebar.toggle();
  }

  protected closeMobile(): void {
    this.sidebar.closeMobile();
  }
}
